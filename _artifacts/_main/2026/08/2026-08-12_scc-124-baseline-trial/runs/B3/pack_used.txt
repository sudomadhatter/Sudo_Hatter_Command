71: def git_root(start: str | Path) -> Path:
72:     """Walk up for the repo root. Running from a subdirectory must not read as 'no hooks'."""
73:     r = subprocess.run(["git", "-C", str(start), "rev-parse", "--show-toplevel"],
74:                        capture_output=True, text=True, errors="replace")
75:     return Path(r.stdout.strip()) if r.returncode == 0 and r.stdout.strip() else Path(start)

78: def _git_config(repo: Path, key: str) -> str:
79:     """`git config --get` exits 1 when the key is unset — that is data, not an error."""
80:     r = subprocess.run(["git", "-C", str(repo), "config", "--get", key],
81:                        capture_output=True, text=True, errors="replace")
82:     return r.stdout.strip() if r.returncode == 0 else ""

85: def _tracked(repo: Path, pathspec: str) -> list[str]:
86:     """Repo-relative paths git actually tracks. The index, not the filesystem — see module doc."""
87:     r = subprocess.run(["git", "-C", str(repo), "ls-files", "-z", "--", pathspec],
88:                        capture_output=True, text=True, errors="replace")
89:     return sorted(p for p in r.stdout.split("\0") if p) if r.returncode == 0 else []

92: def is_executable(p: Path) -> bool:
93:     """⚠ Windows has no POSIX executable bit.
94: 
95:     CPython synthesises `st_mode` there from file attributes and sets the exec bits only for
96:     `.exe/.bat/.cmd/.com`, so every extensionless hook and every `.sh` would read as dead — and
97:     git-for-windows does not consult that bit for hooks anyway. Reporting a running gate as dead
98:     is worse than not checking: it would block close-out on the PC and print `chmod +x` as the
99:     remedy on a machine that has no `chmod`. Existence is the honest answer there.
100:     """
101:     if not p.is_file():
102:         return False
103:     if os.name == "nt":
104:         return True
105:     return p.stat().st_mode & 0o111 != 0

116:         findings.append({"sev": "ERROR", "code": code, "msg": msg})

119:         findings.append({"sev": "WARN", "code": code, "msg": msg})

211: def check(repo: Path, rep) -> dict:
212:     """Fold a scan into a `wf_common.Report`. Used by task_preflight.
213: 
214:     Findings land as ERRORs on purpose. A warn-only arm-check would be the very thing this
215:     script exists to catch: preflight's VERDICT reads "clear to close out and merge" whenever

222:     claim gates — it declares a Jira project, or ships the gate scripts — and yet tracks no
223:     hooks is drift, and drift blocks. A worktree cut before `.githooks/` existed is exactly
224:     that case, and it is completely ungated.
225:     """
226:     res = scan(repo)
227:     for f in res["findings"]:
228:         soft = f.get("code") == "no_hook_dir" and not res["claims_gates"]
229:         (rep.err if f["sev"] == "ERROR" and not soft else rep.warn)("hooks", f["msg"])

231:         rep.info("hooks", f"ARMED - {len(res['hooks'])} hook(s) via core.hooksPath="
232:                           f"{res['hooks_path']}, {len(res['flags'])} arm flag(s) tracked")
233:     return res
